% All CV tables share one fixed-width label column so the content column lines
% up across sections, tables, and pages. They are longtables, so they break
% across pages on their own; end a row with \\* to keep it with the next.
\setlength{\LTleft}{0pt}
\setlength{\LTright}{0pt}
\setlength{\LTpre}{0pt}
\setlength{\LTpost}{0pt}
\newlength{\cvlabelwidth}
\AtBeginDocument{%
  \settowidth{\cvlabelwidth}{\Large Department}%
  \addtolength{\cvlabelwidth}{1em}%
}
\newenvironment{cvtable}
  {\par\normalsize
   \begin{longtable}{@{}p{\cvlabelwidth}@{}p{\dimexpr\textwidth-\cvlabelwidth\relax}@{}}}
  {\end{longtable}}

% Section label for the first row of a cvtable (15pt leading keeps two-line
% labels close to the 12pt row pitch). It is smashed so the larger
% font (and multi-line labels such as Work\\History) do not stretch rows.
\newcommand{\cvlabel}[1]{\smash{\parbox[t]{\cvlabelwidth}{\raggedright\Large\setlength{\baselineskip}{15pt}#1}}}

% Accent-colored entry title (courses, talks, grants, advising groups).
\newcommand{\cventry}[1]{\textcolor{cvaccent}{#1}}

% Indented detail line under an entry; wrapped lines keep the indent.
\newcommand{\cvsub}{\hangindent=1.5em\hangafter=0\relax}

% Link text that keeps the body color, for lists where only some items have
% a URL (e.g. program committees): \href{<url>}{\cvplain{<text>}}. The URL
% must stay a direct \href argument so characters like # survive.
\newcommand{\cvplain}[1]{\textcolor{black}{#1}}

% Publication lists hang their [C1]-style numbers in the label column so
% entries line up with the content column of the cvtables.
\newcommand{\cvpubsection}[1]{%
  \par\vspace{0.8em}\noindent\hspace*{\cvlabelwidth}{\large\scshape #1}\par\nopagebreak}
\newenvironment{cvpubs}[1]
  {\begin{enumerate}[label={[#1\arabic*]}, leftmargin=\cvlabelwidth,
     labelindent=0pt, labelsep=0.6em, labelwidth=!, align=right,
     itemsep=0pt, topsep=0.3em]\emergencystretch=1.5em\relax}
  {\end{enumerate}}
% One entry; pub:<key> resolves to its number (e.g. [C4]) via \ref.
\newcommand{\cvpub}[1]{\item\label{pub:#1}\cvbibentry{#1}}
