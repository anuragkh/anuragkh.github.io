\addbibresource{../../_bibliography/papers.bib}

\newcommand{\cvbiblinebreak}{\\[-0.05\baselineskip]}
\renewcommand*{\mkbibquote}[1]{#1}

\DeclareNameAlias{default}{given-family}
\DeclareFieldFormat{title}{#1}
\DeclareFieldFormat[article]{title}{#1}
\DeclareFieldFormat[inproceedings]{title}{#1}
\DeclareFieldFormat[book]{title}{#1}
\DeclareFieldFormat{usera}{#1}
\DeclareFieldFormat{note}{#1}

\newbibmacro*{cv:title}{%
  \printtext{\textcolor{cvaccent}{\printfield[title]{title}}}}
% Author names: the CV owner in bold, advisees with a dagger. Both are
% registered as given/family pairs (see generated/advisees.tex); a trailing *
% (equal contribution) in the bib is registered as its own variant.
\makeatletter
\newcommand{\cvself}[2]{\csgdef{cvself@\detokenize{#1@#2}}{}}
\newcommand{\cvadvisee}[2]{\csgdef{cvadvisee@\detokenize{#1@#2}}{}}
\DeclareNameFormat{cvauthor}{%
  \nameparts{#1}%
  \edef\cv@namekey{\detokenize\expandafter{\namepartgiven}@\detokenize\expandafter{\namepartfamily}}%
  \ifcsdef{cvself@\cv@namekey}
    {\begingroup
     \renewcommand*{\mkbibnamegiven}[1]{\mkbibbold{##1}}%
     \renewcommand*{\mkbibnamefamily}[1]{\mkbibbold{##1}}%
     \usebibmacro{name:given-family}{\namepartfamily}{\namepartgiven}{\namepartprefix}{\namepartsuffix}%
     \endgroup}
    {\usebibmacro{name:given-family}{\namepartfamily}{\namepartgiven}{\namepartprefix}{\namepartsuffix}%
     \ifcsdef{cvadvisee@\cv@namekey}{\textsuperscript{\dag}}{}}%
  \usebibmacro{name:andothers}}
\makeatother
\newbibmacro*{cv:authors}{%
  \printnames[cvauthor]{author}}
\newbibmacro*{cv:venue+date}{%
  \iffieldundef{booktitle}
    {\printfield{journaltitle}}
    {\printfield{booktitle}}%
  \setunit{\addcomma\space}\printfield{year}}
% Equal-contribution notes are covered by the legend under Publications.
\newbibmacro*{cv:note}{%
  \iffieldequalstr{note}{(*Equal contribution authors)}
    {}
    {\printfield{note}}}
\newbibmacro*{cv:award}{%
  \iffieldundef{usera}
    {}
    {\printtext{\textbf{\printfield{usera}}}}}
\newbibmacro*{cv:entry}{%
  \usebibmacro{cv:title}%
  \ifnameundef{author}
    {}
    {\cvbiblinebreak\usebibmacro{cv:authors}}%
  \iffieldundef{booktitle}
    {\iffieldundef{journaltitle}
      {}
      {\cvbiblinebreak\usebibmacro{cv:venue+date}}}
    {\cvbiblinebreak\usebibmacro{cv:venue+date}}%
  \iffieldundef{note}
    {}
    {\iffieldequalstr{note}{(*Equal contribution authors)}
      {}
      {\cvbiblinebreak\usebibmacro{cv:note}}}%
  \iffieldundef{usera}
    {}
    {\cvbiblinebreak\usebibmacro{cv:award}}}

\DeclareBibliographyDriver{inproceedings}{%
  \usebibmacro{cv:entry}%
  \usebibmacro{finentry}}
\DeclareBibliographyDriver{article}{%
  \usebibmacro{cv:entry}%
  \usebibmacro{finentry}}
\DeclareBibliographyDriver{book}{%
  \usebibmacro{cv:title}%
  \ifnameundef{author}
    {}
    {\cvbiblinebreak\usebibmacro{cv:authors}}%
  \iflistundef{publisher}
    {}
    {\cvbiblinebreak\mkbibemph{PhD Dissertation, \printlist{publisher}}%
     \setunit{\addcomma\space}\printfield{year}}%
  \iffieldundef{note}
    {}
    {\iffieldequalstr{note}{(*Equal contribution authors)}
      {}
      {\cvbiblinebreak\usebibmacro{cv:note}}}%
  \iffieldundef{usera}
    {}
    {\cvbiblinebreak\usebibmacro{cv:award}}%
  \usebibmacro{finentry}}
